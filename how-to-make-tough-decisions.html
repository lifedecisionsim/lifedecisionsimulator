Step 4 & 5: Calculate Scores and Simulate

Compute total scores and simulate decision outcomes:


function calculateScores(options, weights) {
  return options.map(option => {
    const score = 
      option.financial * weights.financial +
      option.workLife * weights.workLife +
      option.growth * weights.growth +
      option.satisfaction * weights.satisfaction;
    return { ...option, totalScore: score };
  });
}

function simulateDecision(option, iterations = 1000) {
  let total = 0;
  for (let i = 0; i < iterations; i++) {
    const randomFactor = Math.random();
    total += option.totalScore * randomFactor;
  }
  return total / iterations;
}

Step 6: Make Your Decision

Pick the option with the highest simulated score. Repeat simulations if needed for confidence.
